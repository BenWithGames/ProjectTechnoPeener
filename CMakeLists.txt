cmake_minimum_required(VERSION 3.22)
project(SC6Rollback LANGUAGES CXX)

option(SC6_BUILD_MOD "Build the UE4SS DLL (requires matching UE4SS sources)" OFF)
set(UE4SS_SOURCE_DIR "" CACHE PATH "Path to initialized RE-UE4SS source checkout")

add_library(sc6_contract INTERFACE)
target_include_directories(sc6_contract INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(sc6_contract INTERFACE cxx_std_20)

if(SC6_BUILD_MOD)
    if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The mod requires Windows x64 and MSVC.")
    endif()
    if(NOT EXISTS "${UE4SS_SOURCE_DIR}/CMakeLists.txt")
        message(FATAL_ERROR "Set UE4SS_SOURCE_DIR to matching RE-UE4SS sources with submodules initialized.")
    endif()
    add_subdirectory("${UE4SS_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/ue4ss")
    enable_language(C ASM_MASM)
    set(MH third_party/minhook)
    add_library(sc6_hook STATIC ${MH}/src/buffer.c ${MH}/src/hook.c
        ${MH}/src/trampoline.c ${MH}/src/hde/hde64.c src/EntryBridge.asm)
    target_include_directories(sc6_hook PUBLIC ${MH}/include)
    add_library(SC6Rollback SHARED src/Mod.cpp src/FrameRecorder.cpp)
    target_link_libraries(SC6Rollback PRIVATE UE4SS sc6_contract sc6_hook bcrypt)
    set_target_properties(SC6Rollback PROPERTIES OUTPUT_NAME main)
    install(TARGETS SC6Rollback RUNTIME DESTINATION SC6Rollback/dlls)
else()
    enable_testing()
    add_executable(sc6_contract_test tests/ContractTest.cpp)
    target_link_libraries(sc6_contract_test PRIVATE sc6_contract)
    add_test(NAME snapshot_replay_contract COMMAND sc6_contract_test)
    add_executable(sc6_session_policy_test tests/SessionPolicyTest.cpp)
    target_link_libraries(sc6_session_policy_test PRIVATE sc6_contract)
    add_test(NAME session_policy COMMAND sc6_session_policy_test)
endif()
